💹 DP on Stocks

State machine DP. At each day: hold, sold, rest states with defined transitions.

State machine DP. At each day: hold, sold, rest states with defined transitions.

⏱ Complexity

Time: O(n) · Space: O(1)

Core Code
// I: 1 transaction
int maxProfit1(int[] prices) {
    int min = Integer.MAX_VALUE, profit = 0;
    for (int p : prices) {
        min = Math.min(min, p);
        profit = Math.max(profit, p - min);
    }
    return profit;
}
// II: unlimited transactions (greedy)
int maxProfit2(int[] prices) {
    int profit = 0;
    for (int i=1;iprices[i-1]) profit+=prices[i]-prices[i-1];
    return profit;
}
// 3 states: hold, sold, rest
// sold → rest (mandatory cooldown)
// rest → hold (can buy)
int maxProfitCooldown(int[] prices) {
    int hold=Integer.MIN_VALUE, sold=0, rest=0;
    for (int p : prices) {
        int prevSold = sold;
        sold = hold + p;
        hold = Math.max(hold, rest - p);
        rest = Math.max(rest, prevSold);
    }
    return Math.max(sold, rest);
}
// 2 states: cash (not holding), hold (holding)
int maxProfitFee(int[] prices, int fee) {
    int cash=0, hold=-prices[0];
    for (int p : prices) {
        cash = Math.max(cash, hold + p - fee); // sell
        hold = Math.max(hold, cash - p);       // buy
    }
    return cash;
}
// dp[k][0] = max profit with at most k txns, not holding
// dp[k][1] = max profit with at most k txns, holding
int maxProfit(int k, int[] prices) {
    if (k >= prices.length/2) return maxProfit2(prices); // unlimited
    int[][] dp = new int[k+1][2];
    for (int[] d : dp) d[1] = Integer.MIN_VALUE;
    for (int p : prices)
        for (int i=k;i>=1;i--) {
            dp[i][0] = Math.max(dp[i][0], dp[i][1]+p);   // sell
            dp[i][1] = Math.max(dp[i][1], dp[i-1][0]-p); // buy
        }
    return dp[k][0];
}
